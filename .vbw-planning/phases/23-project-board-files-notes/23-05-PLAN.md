```ts
/**

export async function updateNotes(cardId: string, notes: string, editorUserId: string) {
```

### Task 2: boardArchiveService — schedule-isolated hard-delete with audit payload
Create `backend/src/services/boardArchiveService.ts`:
```ts
import { prisma } from '../db/prisma.js'
import fs from 'node:fs'
import path from 'node:path'

export class ArchiveError extends Error {
  constructor(public readonly code: 'NOT_FOUND' | 'PROJECT_NAME_MISMATCH' | 'NO_ASSIGNMENT') {
    super(code)
  }
}

export interface ArchiveAuditDetails {
  cardId: string
  projectName: string
  fileCount: number
  totalBytes: number
  adminId: string
}

/**
 * Archive a board card. Validates the typed-confirmation project name, hard-
 * deletes attached files from disk + DB, marks the card archived. Returns audit
 * details for the caller to log.
 *
 * SCHEDULE-ISOLATION INVARIANT (DO NOT REMOVE):
 *   This function MUST NOT call prisma.assignment.*, prisma.teamMember.*,
 *   prisma.absence.*, or prisma.holiday.* — even for reads beyond what is
 *   already on the BoardCard.assignment join shown below. The linked schedule
 *   assignment is read-only here. ANY edit to this function must preserve the
 *   invariant; future reviewers should reject changes that violate it.
 */
export async function archiveCard(cardId: string, confirmProjectName: string, adminUserId: string): Promise<ArchiveAuditDetails> {
  const card = await prisma.boardCard.findUnique({
    where: { id: cardId },
    include: {
      assignment: { select: { projectName: true } }, // READ ONLY — no writes
      files: { select: { id: true, storedName: true, sizeBytes: true } },
    },
  })
  if (!card) throw new ArchiveError('NOT_FOUND')
  if (!card.assignment) throw new ArchiveError('NO_ASSIGNMENT')
  if (card.assignment.projectName !== confirmProjectName) {
    throw new ArchiveError('PROJECT_NAME_MISMATCH')
  }

  const fileCount = card.files.length
  const totalBytes = card.files.reduce((s, f) => s + f.sizeBytes, 0)

  // Disk unlink first (best-effort: missing files don't block archive)
  for (const f of card.files) {
    const onDisk = path.join(process.cwd(), 'uploads', 'board', cardId, f.storedName)
    try { fs.unlinkSync(onDisk) } catch (e) { /* swallow ENOENT */ }
  }

  // Atomic DB cleanup: delete BoardFile rows + mark card archived. Comments and
  // notes stay (CONTEXT.md: "metadata retention").
  await prisma.$transaction([
    prisma.boardFile.deleteMany({ where: { cardId } }),
    prisma.boardCard.update({
      where: { id: cardId },
      data: { archivedAt: new Date(), stage: 'archived' },
    }),
  ])

  return {
    cardId,
    projectName: card.assignment.projectName,
    fileCount,
    totalBytes,
    adminId: adminUserId,
  }
}
```

### Task 3: POST archive route in boardAdmin.ts
In `backend/src/routes/boardAdmin.ts` (mounted at `/cards/:cardId/admin` from plan 01):
```ts
router.post('/archive', requireAuth, requireRole('ADMIN'), mutationRateLimiter, asyncHandler(async (req, res) => {
  const { confirmProjectName } = z.object({ confirmProjectName: z.string().min(1) }).parse(req.body)
  try {
    const details = await archiveCard(req.params.cardId, confirmProjectName, req.session.userId!)
    await logAuditEvent({
      userId: req.session.userId!,
      action: 'board.card.archive',
      ipAddress: extractIp(req),
      details,
    })
    emitBoardInvalidate('cards')
    emitBoardInvalidate('files')
    return res.json({ success: true, ...details })
  } catch (err) {
    if (err instanceof ArchiveError) {
      const status = err.code === 'NOT_FOUND' ? 404 : 400
      return res.status(status).json({ error: err.code })
    }
    throw err
  }
}))
```

Note: `requireRole('ADMIN')` enforces the role-only gate; `requireCardAccess` is NOT used here because admins are universally allowed. Audit event always emitted on success.

**Commit:** `feat(23-05): add POST /cards/:cardId/admin/archive (ADMIN-only, typed-confirmation, audit)`

### Task 4: GET unread-count + POST mark-read in boardNotifications.ts

```ts
router.get('/unread-count', requireAuth, asyncHandler(async (req, res) => {
  const count = await prisma.boardNotification.count({
    where: { userId: req.session.userId!, isRead: false },
  })
  return res.json({ count })
}))

router.post('/mark-read', requireAuth, mutationRateLimiter, asyncHandler(async (req, res) => {
  const { cardId } = z.object({ cardId: z.string().cuid() }).parse(req.body)

    where: { userId: req.session.userId!, cardId, isRead: false },

  })
  emitBoardInvalidate('notifications')
  return res.json({ success: true })
}))
```

`/unread-count` is per-user; admins see only their own count (CONTEXT.md does not call for an admin-aggregated view). `mark-read` is scoped to a single card so opening one card's modal only clears notifications for that card.
